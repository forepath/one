Signed package repositories

apt and rpm repositories signed with the ForePath desktop key

Terminal 
# Debian / Ubuntu
$ curl -fsSL https://downloads.decabill.com/forepath-desktop.asc \
    | sudo gpg --dearmor -o /usr/share/keyrings/forepath-desktop.gpg
$ sudo curl -fsSL -o /etc/apt/sources.list.d/decabill.sources \
    https://downloads.decabill.com/apt/decabill-stable.sources
$ sudo apt update && sudo apt install decabill

# Fedora / RHEL / openSUSE
$ sudo rpm --import https://downloads.decabill.com/forepath-desktop.asc
$ sudo curl -fsSL -o /etc/yum.repos.d/decabill.repo \
    https://downloads.decabill.com/rpm/stable/decabill.repo
$ sudo dnf install decabill
